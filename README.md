```bash
pip install yt-transcript 2>/dev/null || git clone https://github.com/JADRT22/yt-transcript && cd yt-transcript
python -m yt_transcript "https://www.youtube.com/watch?v=dQw4w9WgXcQ" 2>/dev/null || python main.py "https://www.youtube.com/watch?v=dQw4w9WgXcQ"
```
